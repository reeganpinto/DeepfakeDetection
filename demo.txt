import streamlit as st
from tensorflow.keras.models import load_model  # type: ignore
from PIL import Image
import numpy as np
import time

# Load models based on selection
@st.cache_resource
def load_models():
    models = {
        "Custom CNN": load_model("deepfake_custom_model.h5"),
        # Add more models when available, e.g.:
        # "Transfer Learning": load_model("tl_model.keras"),
        # "U-Net": load_model("deepfake_updated_unet_model.h5"),
    }
    return models

models = load_models()

# UI Setup

st.title("🧠 Deepfake Image Detector")
st.markdown("Upload an image to detect whether it's **Real or Fake** using a selected model.")

# Model Selection
model_choice = st.selectbox("Choose a model to use:", list(models.keys()))
model = models[model_choice]

# File Upload
uploaded_file = st.file_uploader("Upload an image (JPG/PNG)", type=["jpg", "jpeg", "png"])

# Image Preprocessing
def preprocess_image(image):
    image = image.convert("RGB")
    image = image.resize((128, 128))  # Match model input size
    img_array = np.array(image) / 255.0
    img_array = np.expand_dims(img_array, axis=0)
    return img_array

# Prediction Logic
if uploaded_file is not None:
    image = Image.open(uploaded_file)
    st.subheader("📸 Uploaded Image")
    st.image(image, caption="Preview", use_container_width=True)

    if st.button("🔍 Predict"):
     with st.spinner("Analyzing image..."):
        start = time.time()

        input_data = preprocess_image(image)
        prediction = model.predict(input_data)[0][0]
        label = "Fake" if prediction >= 0.5 else "Real"
        confidence = prediction if label == "Fake" else 1 - prediction

        # Ensure spinner shows for at least 2 seconds
        elapsed = time.time() - start
        if elapsed < 2:
            time.sleep(2 - elapsed)

        # Display Results
        col1, col2 = st.columns(2)
        with col1:
            st.markdown(f"### Prediction Result:")
            color = "red" if label == "Fake" else "green"
            emoji = "😡" if label == "Fake" else "😊"
            st.markdown(f"<h3 style='color:{color};'>{label} {emoji}</h3>", unsafe_allow_html=True)

        with col2:
            st.metric(label="Confidence", value=f"{confidence:.2%}")
